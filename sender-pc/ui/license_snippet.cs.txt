    // ---------------- 正版注册 ----------------
    static class License
    {
        const string Secret = "XL-2026-PROJECTOR-SIGNKEY-7f3a91c4";

        /// <summary>机器码：机器名 + 系统盘卷序列号 + CPU 核数，取 SHA256 前 12 位十六进制。</summary>
        public static string MachineCode()
        {
            try
            {
                string vol = "0";
                try
                {
                    var d = new System.IO.DriveInfo(Path.GetPathRoot(Environment.SystemDirectory));
                    if (d != null) vol = d.IsReady ? d.VolumeLabel + ":" + d.TotalSize : "0";
                }
                catch { }
                string raw = Environment.MachineName + "|" + vol + "|" + Environment.ProcessorCount;
                using (var sha = System.Security.Cryptography.SHA256.Create())
                {
                    byte[] h = sha.ComputeHash(System.Text.Encoding.UTF8.GetBytes(raw));
                    var sb = new System.Text.StringBuilder();
                    for (int i = 0; i < 6; i++) sb.Append(h[i].ToString("X2"));
                    string hex = sb.ToString();
                    return hex.Substring(0, 4) + "-" + hex.Substring(4, 4) + "-" + hex.Substring(8, 4);
                }
            }
            catch { return "0000-0000-0000"; }
        }

        /// <summary>由机器码推导出的注册码（发行方用同样算法生成，见 tools\makelicense.ps1）。</summary>
        public static string ExpectedKey(string machineCode)
        {
            using (var sha = System.Security.Cryptography.SHA256.Create())
            {
                byte[] h = sha.ComputeHash(System.Text.Encoding.UTF8.GetBytes(Secret + "|" + machineCode));
                var sb = new System.Text.StringBuilder();
                for (int i = 0; i < 6; i++) sb.Append(h[i].ToString("X2"));
                string hex = sb.ToString();
                return "XL-" + hex.Substring(0, 4) + "-" + hex.Substring(4, 4) + "-" + hex.Substring(8, 4);
            }
        }

        static string StorePath
        {
            get
            {
                string dir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "星联投屏");
                Directory.CreateDirectory(dir);
                return Path.Combine(dir, "license.dat");
            }
        }

        public static string SavedKey()
        {
            try { return File.Exists(StorePath) ? File.ReadAllText(StorePath).Trim() : ""; }
            catch { return ""; }
        }

        public static bool IsRegistered()
        {
            string k = SavedKey();
            return k.Length > 0 && k.Equals(ExpectedKey(MachineCode()), StringComparison.OrdinalIgnoreCase);
        }

        public static bool Register(string key)
        {
            key = (key ?? "").Trim().ToUpperInvariant();
            if (!key.Equals(ExpectedKey(MachineCode()), StringComparison.OrdinalIgnoreCase)) return false;
            try { File.WriteAllText(StorePath, key); return true; }
            catch { return false; }
        }
    }
